misc.env-i @@ env -i A=1 env
misc.basename @@ basename /a/b.txt .txt; dirname /a/b.txt
misc.realpath @@ cd /; realpath .
misc.timeout @@ timeout 1 sleep 0.01; echo $?
misc.nl-in-dq @@ echo "a
b"
misc.line-continuation-cmd @@ echo a \
  b
misc.long-and-or @@ true && false || echo rescued
misc.negate-pipeline @@ ! true | false; echo $?
misc.empty-cmd @@ ;; 2>&1 | head -1
misc.semicolon-after-amp @@ sleep 0 & echo ok; wait
